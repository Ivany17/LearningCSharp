Тема №35: Робота зі списками об'єктів (Lists of Objects) 👥📦
Ми навчилися створювати список рядків (string). Але справжня потужність C# проявляється тоді, 
коли ми створюємо список власних об'єктів. Наприклад, список усіх гравців на сервері.

⚠️ Як це працює:
Замість string у кутових дужках ми пишемо назву нашого класу:
List<Player> party = new List<Player>();
‼️ Тепер у цей список можна додавати цілих персонажів з їхніми іменами та рівнями.

Головна відмінність від List<string>:
Коли ми створюємо список List<Vehicle>, ми створюємо список посилань.
Типізація: Ми можемо класти в список будь-який об'єкт, який є "дитиною" Vehicle.
Доступ: Через цикл foreach (var v in garage) ми маємо доступ до всіх публічних методів та властивостей, які є в базовому класі Vehicle.

class Vehicle
{
    public string Brand { get; private set; }
    public int Speed { get; private set; }
    public static int VehicleCount = 0; // static
    public Vehicle(string Brand, int Speed)
    {
        this.Brand = Brand;
        this.Speed = Speed;
        VehicleCount++; //static
    }
    public void Drive()
    {
        Console.WriteLine($"Транспорт {Brand} їде зі швидкістю {Speed} км/год");
    } 
}

class Car : Vehicle
{
    private int Fuel;
    public int FuelLevel 
    { 
        get { return Fuel; } // Дозволяємо читати приватне поле
    }
    public Car(string Brand, int Speed, int Fuel) : base(Brand, Speed)
    {
        this.Fuel = Fuel;
    }
    public void ShowFuel()
    {
        Console.WriteLine($"Палива залишилось: {Fuel}");
    }
    public void Refuel(int amount)
    {
        if(amount > 0)
        {
            Fuel += amount;
            Console.WriteLine($"Машина {Brand} заправлена до {Fuel}");
        } else
        {
            Console.WriteLine("Помилка! Неможливо заправити від'ємну кількість");
        }
    }
}

class ElectricCar : Vehicle
{
    public int BatteryLevel { get; private set; }
    public ElectricCar(string Brand, int Speed, int BatteryLevel) : base(Brand, Speed)
    {
        this.BatteryLevel = BatteryLevel;
    }
}

using System.Collections.Generic;
List<Vehicle> garage = new List<Vehicle>();
garage.Add(new Car("BMW", 230, 45));
garage.Add(new Car("Audi", 210, 52));
garage.Add(new ElectricCar("Tesla", 190, 78));
garage.Add(new ElectricCar("Nissan Leaf", 160, 41));
garage.RemoveAt(3); // оскільки після видалення список стискається, то потрібно видаляти з кінця
garage.RemoveAt(2);

foreach(var car in garage)
{
    car.Drive();
    Vehicle.TotalSpeed += car.Speed;
}
Console.WriteLine($"Середня швидкість усіх машин в гаражі: {Vehicle.TotalSpeed / Vehicle.VehicleCount} км/год");

Console.WriteLine("--- Архів усіх створених моделей ---");

class Vehicle
{
    public string Brand { get; private set; }
    public int Speed { get; private set; }
    public static double TotalSpeed = 0; // static
    public static int VehicleCount = 0; // static
    public static List<string> AllVehicleModels = new List<string>(); // static
    public Vehicle(string Brand, int Speed)
    {
        this.Brand = Brand;
        this.Speed = Speed;
        VehicleCount++; //static
        AllVehicleModels.Add(Brand);
    }
    public void Drive()
    {
        Console.WriteLine($"Транспорт {Brand} їде зі швидкістю {Speed} км/год");
    } 
}
